Core Implementation - transcript_service.py 
# Resilient multi-method extraction with graceful degradation
class TranscriptService:
    async def extract_transcript(self, video_id: str) -> Tuple[bool, str]:
        methods = [
            self._youtube_transcript_api,    # Primary: Official API
            self._ytdlp_subtitles,          # Secondary: Direct extraction
            self._manual_caption_fetch,      # Tertiary: HTML parsing
            self._alternative_sources        # Final: Creative extraction
        ]
        
        for i, method in enumerate(methods, 1):
            try:
                success, transcript = await method(video_id)
                if success and transcript:
                    await self._log_extraction_success(method.__name__, i)
                    return True, transcript
            except Exception as e:
                await self._log_extraction_failure(method.__name__, e)
                continue
        
        # Graceful degradation - return empty transcript for chat-only mode
        return False, ""

Context-Aware AI Integration

Complex Integration

The Challenge

Token Limitations: GPT-4 context windows (128k tokens) vs. full video transcripts (often 200k+ tokens)

Context Preservation: Maintaining conversation relevance with video content while handling multi-turn conversations

Performance Constraints: Balancing context richness with response speed and API costs

Strategic Approach: Smart Context Management

  • Full Context Injection: Entire transcript as system context for accurate responses
  • Async Processing: Non-blocking OpenAI API integration with connection pooling
  • Smart Prompting: Structured prompts with video metadata and user query context
  • Error Resilience: Graceful degradation on API failures with retry logic
  • Cost Optimization: Token counting and intelligent context truncation

# Smart context management with token optimization
async def get_ai_response(video_id: str, user_message: str) -> str:
    video = await get_video_by_id(video_id)
    if not video or not video.transcript:
        return "No video context available for this conversation."
    
    # Token-aware context preparation
    context_tokens = count_tokens(video.transcript)
    max_context = 100000  # Reserve tokens for response
    
    if context_tokens > max_context:
        # Intelligent truncation preserving key content
        transcript = smart_truncate(video.transcript, max_context)
    else:
        transcript = video.transcript
    
    messages = [
        {
            "role": "system", 
            "content": f"""You are an AI assistant analyzing video content.
            Video Title: {video.title}
            Duration: {video.duration}
            Transcript: {transcript}
            
            Provide accurate, helpful responses based ONLY on the video content."""
        },
        {"role": "user", "content": user_message}
    ]
    
    # Async API call with retry logic
    return await openai_client.chat.completions.create(
        model="gpt-4-turbo",
        messages=messages,
        max_tokens=1000,
        temperature=0.3
    )
